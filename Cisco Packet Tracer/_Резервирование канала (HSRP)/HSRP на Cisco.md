Поднимаем HSRP (резервирование канала) на роутерах Cisco.  
_________________________________________________________________________________________________________________  

R1 (первый левый роутер, Cisco 2911).  

en  
conf t  

int g0/0 // Заходим на наш внешний интерфейс.  
ip address 10.10.10.1 255.255.255.252 // Вешаем на него IP-адрес и 30-ю маску подсети.  
no sh // И поднимаем.  

int g0/1 // Это интерфейс, который смотрит в нашу внутреннюю сеть, т.е. в сторону коммутатора.  
no sh // Поднимаем и его.  

interface g0/1.10 // Рисуем сабинтерфейс на 10 VLAN.  
encapsulation dot1Q 10  
ip address 192.168.10.1 255.255.255.0  
standby 10 ip 192.168.10.254 // standby 10 - это группа HSRP, 192.168.10.254 - это IP-адрес нашего виртуального шлюза по умолчанию.  
standby 10 priority 110 // Задаём приоритет. Чем выше число, тем главнее роутер.  
standby 10 preempt // Это значит, что если роутер выключился, то после включения, он снова становится главным.  

// Дальше всё тоже самое, но на сабинтерфейсе g0/1.20  
interface g0/1.20  
encapsulation dot1Q 20  
ip address 192.168.20.1 255.255.255.0  
standby 10 ip 192.168.20.254  
standby 10 priority 110  
standby 10 preempt  

// Теперь создаём DHCP-пулы на R1.  
ip dhcp pool 10 // Первый пул на 10 VLAN.  
network 192.168.10.0 255.255.255.0 // Это 10 сетка по 24 маске.  
default-router 192.168.10.254 // Это шлюз по умолчанию.  
ip dhcp pool 20 // Это пул на 20 VLAN.  
network 192.168.20.0 255.255.255.0 // Это 20 сетка по 24 маске.  
default-router 192.168.20.254 // Это также шлюз по умолчанию.  
exit  
ip dhcp excluded-address 192.168.10.1 192.168.10.6 // Это исключения на раздатку первых 5 адресов для 10 VLAN-a.  
ip dhcp excluded-address 192.168.20.1 192.168.20.6 // Это исключения на раздатку первых 5 адресов для 20 VLAN-a.  

// Теперь настроим OSPF на R1.  
router ospf 1  
network 192.168.10.0 0.0.0.255 area 0  
network 192.168.20.0 0.0.0.255 area 0  
network 10.10.10.0 0.0.0.3 area 0  

exit  

// Сохраняем конфиг на R1 (Cisco 2911).  
do wr

R2 (второй роутер, Cisco 2911).  

en  
conf t  

int g0/0 // Заходим на наш внешний интерфейс g0/0.  
ip address 20.20.20.1 255.255.255.252 // Прописываем ему IP-адрес и маску подсети (префикс по CIDR: 30).  
no sh // И поднимаем.  

int g0/2 // Далее заходим на интерфейс, который смотрит во внутреннюю сеть, в сторону коммутатора.  
no sh // Поднимаем его.  

// Далее настраиваем сабинтерфейс.  
interface g0/2.10 // Заходим в сам сабинтерфейс.  
encapsulation dot1Q 10 // Включаем инкапсуляцию, 10 VLAN.  
ip address 192.168.10.2 255.255.255.0 // Прописываем IP-адрес и маску.  
standby 10 ip 192.168.10.254 // Прописываем виртуальный адрес.  
standby 10 priority 100 // Приоритет ставим меньше, чтобы он был запасным.  
standby 10 preempt // Чтобы включился после отключения.  

// Всё тоже самое делаем на втором сабинтерфейсе.  
interface g0/2.20  
encapsulation dot1Q 20  
ip address 192.168.20.2 255.255.255.0  
standby 10 ip 192.168.20.254  
standby 10 priority 100  
standby 10 preempt  

exit  

// Теперь создаём DHCP-пулы на R2.  
ip dhcp pool 10  
network 192.168.10.0 255.255.255.0  
default-router 192.168.10.254  
ip dhcp pool 20  
network 192.168.20.0 255.255.255.0  
default-router 192.168.20.254  
exit  
ip dhcp excluded-address 192.168.10.1 192.168.10.6  
ip dhcp excluded-address 192.168.20.1 192.168.20.6  

// Теперь настроим OSPF на R2.  
router ospf 1  
network 192.168.10.0 0.0.0.255 area 0  
network 192.168.20.0 0.0.0.255 area 0  
network 20.20.20.0 0.0.0.3 area 0  

exit  

// Сохраняем конфиг на R2 (Cisco 2911).  
do wr

Теперь настраиваем коммутатор Cisco 2960, SW1.  

en  
conf t  

int range g0/1-2  
switchport mode trunk  

int f0/1  
switchport mode access  
switchport access vlan 10  

int f0/2  
switchport mode access  
switchport access vlan 20  

exit  

do wr

R3 (первый правый роутер, Cisco 2911).  

en  
conf t  

int g0/0 // Заходим на внешний интерфейс.

default-router 192.168.30.254

network 192.168.40.0 255.255.255.0  
default-router 192.168.40.254
