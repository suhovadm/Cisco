// Как настроить нормальный шрифт, как в настоящей консоли Cisco?

// 1. Настраиваем первый роутер слева, 2911.  

Would you like to enter the initial configuration dialog? [yes/no]: n  
И пару раз Enter  

en // включаем заходим в привилегированный режим  
conf t // переходим в режим конфигурации  

int g0/1 // заходим на интерфейс g0/1  
// это интерфейс, который смотрит в сторону второго роутера. Появится обозначение config-if  

ip address 10.10.10.1 255.255.255.252 // вешаем на него ip-шник и маску  
// маска 30-я, в которой всего 4 адреса. Этого достаточно в нашем случае  

no sh // поднимаем наш интерфейс, потому что он выключен  
// Появится сообщение: change state to up  

int g0/0 // теперь заходим на интерфейс g0/0  
// это интерфейс, который смотрит в сторону коммутатора.  
// Нам надо будет разделить его на 2 сабинтерфейса, т.к. у нас будет 2 шлюза по умолчанию,  
// т.к. у нас будет 2 VLAN-a. Один интерфейс не может быть сразу 2 шлюзами по умолчанию.  

no sh // включаем g0/0.  
// Здесь тоже появится change state to up.  

int g0/0.10 // переходим на первый сабинтерфейс.  

encapsulation dot1Q 10 // и прописываем ему 10 VLAN.  

ip address 192.168.10.1 255.255.255.0 // указываем 10 VLAN-у IP-адрес.  

no sh // и на всякий случай ещё разок поднимаем, хотя он уже будет поднят, скорее всего.  

int g0/0.20 // переходим на второй сабинтерфейс.  

encapsulation dot1Q 20 // указываем ему 20 VLAN.  

ip address 192.168.20.1 255.255.255.0 // указываем 20 VLAN-у IP-адрес.  

no sh // и на всякий случай поднимаем.  

ex // выходим.  

// Теперь настроим DHCP-pool. Компьютер будет получать адреса автоматически,  
// а сервер будет на статике. Сервера и МФУ-шки обычно ставят на статику.  

ip dhcp pool 10 // формируем пул IP-адресов. После этой команды появится (dhcp-config).  
network 192.168.10.0 255.255.255.0 // пишем сетку.  
default-router 192.168.10.1 // указываем шлюз по умолчанию.  
dns-server 192.168.20.4 // это адрес самого сервера.  
// Мы его настроим попозже и поставим ему этот статический адрес.  

exit // выходим.  

// Теперь настроим динамическую маршрутизацию между сетями, т.е. OSPF.  

router ospf 6161 // 6161 - это id нашего процесса. Здесь можно поставить любое число...  
// Обратите внимание, что после этого мы попадём в раздел конфигурирования, который  
// называется (config-router)#  

network 192.168.10.0 0.0.0.255 area 0  
// Здесь мы указываем одну из сетей роутера, обратную маску подсети и area.  
// area везде должна быть одинаковой, чтобы роутеры могли связываться между собой.  

network 192.168.20.0 0.0.0.255 area 0  
// Не забываем указать вторую сеть нашего роутера.  
// Здесь всё тоже самое, только сетка уже не 10, а 20. Все остальные параметры те же самые.  

network 10.10.10.0 0.0.0.3 area 0  
// Это третья сеть, которая у нас между роутерами.  

end  

copy running-config startup-config  
// Если выскочит вопрос "Destination filename [startup-config]? просто жмахаем Enter!  

_______________________________________________________________________________________________  

// 2. Теперь настроим коммутатор слева, 2960.  

en  
conf t  
int f0/1 // заходим на тот интерфейс, который смотрит в сторону нашего роутера.  
// Как только мы введём эту команду, появится сообщение (config-if).  

switchport mode trunk // переводим этот порт в режим транка.  
// Транк всегда смотрит в сторону роутера.  
// Аксессы смотрят в сторону хостов. Это, чтобы коммутатор понимал откуда куда идут данные.  

int f0/2 // заходим на интерфейс f0/2.  

switchport mode access // переводим его в режим аксесс, т.к. к нему подключён комп.  

switchport access vlan 10 // накидываем аксесс порт на 10 VLAN.  
// Если выскочит % Access VLAN does not exist. Creating vlan 10 -  
// просто ещё раз повторяем эту команду.  

int f0/3 // заходим на интерфейс f0/3.  

switchport mode access  

switchport access vlan 20  

Кликаем на ПК - Desktop - IP Configuration -  
переключаемся на DHCP и смотрим, что он поймал параметры.  

На сервере почти тоже самое - Desktop - IP Configuration -  
ставим Static -  
IP: 192.168.20.4  
Mask: 255.255.255.0  
Gateway: 192.168.20.1  
DNS: 192.168.20.4  
Теперь заходим в Services - DNS - ON  
Прописываем любое имя латиницей, address: 192.168.20.4 - add.  
Всё, DNS-сервер поднят.  
Первая сеть настроена!  

end  
copy running-config startup-config  
enter  

_______________________________________________________________________________________________  

// 3. Настраиваем второй роутер справа, 2911.  

Would you like to enter the initial configuration dialog? [yes/no]: n  
И пару раз Enter  

en  
conf t  

int g0/1 // заходим на интерфейс, который смотрит в сторону первого роутера.  

ip address 10.10.10.2 255.255.255.252 // прописываем ему IP-шник и маленьку маску на 4 адреса.  

no sh // поднимаем!  

int g0/0 // заходим на интерфейс, который смотрит в сторону коммутатора.  

no sh // поднимаем и его!  

int g0/0.30 // рисуем на нём 30 VLAN  

encapsulation dot1Q 30  

ip address 192.168.30.1 255.255.255.0  

no sh  

int g0/0.40 // рисуем 40 VLAN  

encapsulation dot1Q 40  

ip address 192.168.40.1 255.255.255.0  

no sh  

ex // выходим.  

// Теперь настроим DHCP, здесь уже будет 2 DHCP-пула, т.к. 2 компа в разных VLAN-ах.  

// Первый пул.  
ip dhcp pool 30  
network 192.168.30.0 255.255.255.0  
default-router 192.168.30.1  
dns-server 192.168.20.4  

// Второй пул.  
ip dhcp pool 40  
network 192.168.40.0 255.255.255.0  
default-router 192.168.40.1  
dns-server 192.168.20.4  

// Теперь настроим OSPF.  

router ospf 54841 // 54841 - это id процесса, он может быть любым.  

network 192.168.30.0 0.0.0.255 area 0 // здесь мы указываем те сети, которые живут СПРАВА, а не к которым мы будем подключаться!  

network 192.168.40.0 0.0.0.255 area 0  

network 10.10.10.0 0.0.0.3 area 0 // маленькая сетка посерёдке, для связанности сетей.  

// На этом моменте должно появиться сообщение, что OSPF увидел соседа, а значит всё круто.  
// Сообщение может выскочить не сразу, а с некоторой задержкой, это нормально.  
// Оно выглядит как-то так:  

00:37:23: %OSPF-5-ADJCHG: Process 54841, Nbr 192.168.20.1 on GigabitEthernet0/1 from LOADING to FULL, Loading Done  

end  
copy running-config startup-config  
enter  

_______________________________________________________________________________________________  

// 4. Настроим второй коммутатор справа, 2960.  

en  
conf t  

int f0/1 // заходим на интерфейс, который смотрит в сторону роутера.  

switchport mode trunk // переводим его в режим транка.  

int f0/2 // теперь заходим на интерфейс f0/2.  

switchport mode access // переводим его в режим аксесса.  

switchport access vlan 40 // закидываем порт в 40 vlan (если заартачится - вбиваем команду ещё раз).  

int f0/3 // заходим на интерфейс f0/3.  

switchport mode access // переводим в режим аксесса.  

switchport access vlan 30 // закидываем порт в 30 vlan (если заартачится - вбиваем команду ещё раз).  

end  
copy running-config startup-config  
enter
